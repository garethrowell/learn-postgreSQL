JSON data type
--------------

Postgresql has native json data type
  json as text
  json as binary and indexable;

want to display titles and tags pairs

select p.pk, p.title, t.tag from posts p
  left join j_posts_tags jpt on p.pk = jpt.post_pk
  left join tags t on jpt.tab_pk = t.pk
  order by 1;

this displays our title/tag pairs including repeats

use 'group by' to elimnate repeats and aggregate data

select p.pk, p.title, string_agg(t.tag, ',') as tag
  from posts p
  left join j_posts_tags jpt on p.pk = jpt.post_pk
  left join tags t on jpt.tag_pk = t.pk
  group by 1, 2
  order by 1;

generate the json structure...

select row_to_json(q) as json_data from(
  select p.pk, p.title, string_agg(t.tag, ',') as tag
    from posts p
    left join j_posts_tags jpt on p.pk = jpt.post_pk
    left join tags t on jpt.tag_pk = t.pk
    group by 1, 2
    order by 1;
  ) Q;

now, call the table post_json with only one jsonb field
  called jsondata

create empty table post_json(json_data jsonb);

insert data into post_json(json_data)(
  select row_to_json(q) as json_data from(
    select p.pk, p.title, string_agg(t.tag, ',') as tag
      from posts p
      left join j_posts_tags jpt on p.pk = jpt.post_pk
      left join tags t on jpt.tag_pk = t.pk
      group by 1, 2
      order by 1;
   ) Q
)

now to view the new jsonb table...

select jsonb_pretty(jsondata) from post_json;

or doing a where clause

select jsonb_pretty(jsondata) from post_json
  where jsondata @> { "tag":"Database"}

@> operator checks where left JSON value contains the
  right path/value entries;

Polymorphic SQL functions
-------------------------

here, we want to write a single function that can
use all types of data
likened to the Oracle NVL function which accepts 
NULL parameters

create or replace function nvl (anyelement, anyelement)
  returns anyelement as $$
  select coalesce ($1, $2);
  $$
  language SQL;
  create function

making calls to a polymorphic function

select nvl(''::text, 'n'::text);

select nvl('a'::text, 'n'::text);

-----------------

create or replace function my_sum_3_params
  (IN x integer, IN y integer, OUT z integer)
  AS

  BEGIN
    z: = x + y;
  END;
  $BODY$
  language 'plpgsql';
  CREATE FUNCTION

select my_sum_3_params(2,3);

IN ~ input paramters (default)
OUT ~ output parameters
INOUT ~ input / output paramters

Functions are VOLATILE (default)
              STABLE
	      IMMUTABLE

VOLATILE ~ can change the database (most common)
STABLE ~ cannot modify the database
IMMUTABLE ~ always returns same result 
            (optimizer pre-evaluates)

examples...

now() is a STABLE function
  value is fixed w/i transaction
  no change to the database

lower(string-expression) is an immutable function
  returns same result across multiple transactions

Exception handling statements
-----------------------------

example catching division by 0

create or replace function my_second_except(x real, y real)
  returns real as
  $$
  DECLARE
    ret real;
  BEGIN
    ret := x/y;
    return ret;
  EXCEPTION
    WHEN division_by_zero THEN
      RAISE INFO 'DIVISION BY ZERO';
      RAISE INFO 'Error % %,
        SQLSTATE, SQLERRM;
      RETURN 0;
  END;
  $$
  language 'plpgsql';
  CREATE FUNCTION

Security definer
----------------

invoke a function as if you are the owner
  typically to access system info from a regular user

create function forum.my_stat_activity()
  returns table (pid integer, query text)
  as $$
    select p.id, query from pg_stat_activity;
  $$ language 'sql'
  security definer;  -- modify security here...
  grant execute on function forum.my_stat_activity
    to forum;

  select * from my_stat_activity -- run as forum
